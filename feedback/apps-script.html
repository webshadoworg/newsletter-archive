/**
 * איסוף משובים מדף הנחיתה של העלון -> Google Sheets
 * ------------------------------------------------------------------
 * נוצר 20.8.2026 עבור רפאל. הקוד הזה נכנס ל-Apps Script של גוגל שיטס
 * (תוספים > Apps Script), נפרס כ-Web app, וה-URL שמתקבל נכנס למשתנה
 * ENDPOINT בקובץ hebrew-newsletter/feedback/index.html.
 *
 * הדף שולח *פעמיים* עם אותו d.id:
 *   1. בטעינה  - {id,g,p,s,r}        הדירוג בכוכבים
 *   2. בשליחה  - {id,g,p,s,c,e}      ההערה החופשית + מייל (אם סומן)
 * לכן הסקריפט עושה upsert לפי id: שורה אחת לכל משוב, לא שתיים.
 * שדה ריק בשליחה השנייה לא מוחק ערך שכבר נשמר.
 *
 * הדף שולח ב-mode:"no-cors" עם Content-Type: text/plain - בקשה "פשוטה"
 * שלא דורשת preflight, ולכן היא עוברת ל-Apps Script בלי CORS.
 * המשמעות: הדף לא רואה את התשובה. אין טעם להחזיר שגיאות.
 */

var SHEET = 'משובים';
var NAMES = { '1': 'המאור שבה', '2': 'מתוק לנפש', '3': 'צעדני ואושעה', '4': 'צידה לדרך' };
var HEAD = ['תאריך', 'מזהה', 'גיליון', 'פרשה', 'מדור', 'שם המדור', 'דירוג', 'הערה', 'מייל'];

function sheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SHEET) || ss.insertSheet(SHEET);
  if (sh.getLastRow() === 0) {
    sh.appendRow(HEAD);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, HEAD.length).setFontWeight('bold');
    sh.setRightToLeft(true);
  }
  return sh;
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var d = JSON.parse(e.postData.contents);
    var sh = sheet_();
    var vals = [new Date(), d.id || '', d.g || '', d.p || '', d.s || '',
                NAMES[d.s] || '', d.r || '', d.c || '', d.e || ''];

    var last = sh.getLastRow();
    var row = 0;
    if (last > 1 && d.id) {
      var ids = sh.getRange(2, 2, last - 1, 1).getValues();
      for (var i = 0; i < ids.length; i++) {
        if (String(ids[i][0]) === String(d.id)) { row = i + 2; break; }
      }
    }

    if (row) {
      var cur = sh.getRange(row, 1, 1, HEAD.length).getValues()[0];
      vals[0] = cur[0];                                  // לשמור על זמן הפנייה הראשונה
      for (var j = 6; j <= 8; j++) if (!vals[j]) vals[j] = cur[j];
      sh.getRange(row, 1, 1, HEAD.length).setValues([vals]);
    } else {
      sh.appendRow(vals);
    }
  } catch (err) {
    Logger.log(err);
  } finally {
    lock.releaseLock();
  }
  return ContentService.createTextOutput('ok');
}

function doGet() {
  return ContentService.createTextOutput('ok');   // בדיקת חיים בדפדפן
}
