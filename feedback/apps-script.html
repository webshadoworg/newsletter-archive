/**
 * GYE weekly newsletter - feedback collector (Google Sheets / Apps Script)
 * ------------------------------------------------------------------------
 * 20.8.2026 - ASCII-ONLY BUILD. Every Hebrew string is written as \uXXXX
 * escapes and there is not a single non-ASCII byte in this file, so it cannot
 * be corrupted by an email client, a smart-quote autocorrect, or an injected
 * bidi control character. Paste this one through ANY channel.
 * The readable Hebrew original lives at hebrew-newsletter/feedback/apps-script.gs
 * - edit that one, then regenerate this.
 *
 * The landing page POSTs TWICE with the same d.id:
 *   1. on load   {id,g,p,s,r}     the star rating
 *   2. on submit {id,g,p,s,c,e}   free-text comment + email if opted in
 * So we upsert by id: one row per reader per section, never two half-empty rows.
 * An empty field in the second POST never wipes a value already stored.
 *
 * The page posts with mode:"no-cors" and Content-Type text/plain - a "simple"
 * request, no preflight, so it reaches Apps Script without CORS. Consequence:
 * the page never sees the response. Returning errors is pointless.
 */

var SHEET = '\u05de\u05e9\u05d5\u05d1\u05d9\u05dd';

var NAMES = {
  '1': '\u05d4\u05de\u05d0\u05d5\u05e8 \u05e9\u05d1\u05d4',
  '2': '\u05de\u05ea\u05d5\u05e7 \u05dc\u05e0\u05e4\u05e9',
  '3': '\u05e6\u05e2\u05d3\u05e0\u05d9 \u05d5\u05d0\u05d5\u05e9\u05e2\u05d4',
  '4': '\u05e6\u05d9\u05d3\u05d4 \u05dc\u05d3\u05e8\u05da'
};

var HEAD = [
  '\u05ea\u05d0\u05e8\u05d9\u05da',
  '\u05de\u05d6\u05d4\u05d4',
  '\u05d2\u05d9\u05dc\u05d9\u05d5\u05df',
  '\u05e4\u05e8\u05e9\u05d4',
  '\u05de\u05d3\u05d5\u05e8',
  '\u05e9\u05dd \u05d4\u05de\u05d3\u05d5\u05e8',
  '\u05d3\u05d9\u05e8\u05d5\u05d2',
  '\u05d4\u05e2\u05e8\u05d4',
  '\u05de\u05d9\u05d9\u05dc'
];

function sheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SHEET) || ss.insertSheet(SHEET);
  if (sh.getLastRow() === 0) {
    sh.appendRow(HEAD);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, HEAD.length).setFontWeight('bold');
    sh.setRightToLeft(true);
  }
  return sh;
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var d = JSON.parse(e.postData.contents);
    var sh = sheet_();
    var vals = [new Date(), d.id || '', d.g || '', d.p || '', d.s || '',
                NAMES[d.s] || '', d.r || '', d.c || '', d.e || ''];

    var last = sh.getLastRow();
    var row = 0;
    if (last > 1 && d.id) {
      var ids = sh.getRange(2, 2, last - 1, 1).getValues();
      for (var i = 0; i < ids.length; i++) {
        if (String(ids[i][0]) === String(d.id)) { row = i + 2; break; }
      }
    }

    if (row) {
      var cur = sh.getRange(row, 1, 1, HEAD.length).getValues()[0];
      vals[0] = cur[0]; // keep the first-contact timestamp
      for (var j = 6; j <= 8; j++) if (!vals[j]) vals[j] = cur[j];
      sh.getRange(row, 1, 1, HEAD.length).setValues([vals]);
    } else {
      sh.appendRow(vals);
    }
  } catch (err) {
    Logger.log(err);
  } finally {
    lock.releaseLock();
  }
  return ContentService.createTextOutput('ok');
}

function doGet() {
  return ContentService.createTextOutput('ok'); // liveness check in a browser
}
