function sheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SHEET) || ss.insertSheet(SHEET);
  if (sh.getLastRow() === 0) {
    sh.appendRow(HEAD);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, HEAD.length).setFontWeight('bold');
    sh.setRightToLeft(true);
  } else if (sh.getLastColumn() < HEAD.length) {
    for (var i = sh.getLastColumn(); i < HEAD.length; i++) {
      sh.getRange(1, i + 1).setValue(HEAD[i]).setFontWeight('bold');
    }
  }
  return sh;
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var d = JSON.parse(e.postData.contents);
    var sh = sheet_();
    var vals = [new Date(), d.id || '', d.g || '', d.p || '', d.s || '',
                NAMES[d.s] || '', d.r || '', d.c || '', d.e || '', d.rid || ''];

    var last = sh.getLastRow();
    var row = 0;
    if (last > 1 && d.id) {
      var ids = sh.getRange(2, 2, last - 1, 1).getValues();
      for (var i = 0; i < ids.length; i++) {
        if (String(ids[i][0]) === String(d.id)) { row = i + 2; break; }
      }
    }

    if (row) {
      var cur = sh.getRange(row, 1, 1, HEAD.length).getValues()[0];
      vals[0] = cur[0];
      for (var j = 6; j <= 9; j++) if (!vals[j]) vals[j] = cur[j];
      sh.getRange(row, 1, 1, HEAD.length).setValues([vals]);
    } else {
      sh.appendRow(vals);
    }
  } catch (err) {
    Logger.log(err);
  } finally {
    lock.releaseLock();
  }
  return ContentService.createTextOutput('ok');
}

function doGet() {
  return ContentService.createTextOutput('ok');
}

function setupSummary() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SUMMARY) || ss.insertSheet(SUMMARY);
  sh.clear();
  sh.setRightToLeft(true);

  var Q = "'" + SHEET + "'!A:I";
  var q = function (sel) { return '=QUERY(' + Q + ',"' + sel + '",1)'; };
  var lbl = function (col, text) { return col + " '" + text + "'"; };

  var f1 = q('select F, avg(G), count(G) where G is not null and F is not null ' +
             'group by F order by avg(G) desc label ' +
             lbl('F', L.sec) + ', ' + lbl('avg(G)', L.avg) + ', ' + lbl('count(G)', L.cnt));

  var f2 = q('select C, D, avg(G), count(G) where G is not null ' +
             'group by C, D order by C label ' +
             lbl('C', L.issue) + ', ' + lbl('D', L.parsha) + ', ' +
             lbl('avg(G)', L.avg) + ', ' + lbl('count(G)', L.cnt));

  var f3 = q('select C, avg(G) where G is not null group by C pivot F label ' + lbl('C', L.issue));

  var blocks = [[3, L.t1, f1], [12, L.t2, f2], [24, L.t3, f3]];
  for (var i = 0; i < blocks.length; i++) {
    sh.getRange(blocks[i][0], 1).setValue(blocks[i][1]).setFontWeight('bold').setFontSize(12);
    sh.getRange(blocks[i][0] + 1, 1).setFormula(blocks[i][2]);
  }

  sh.getRange(1, 1).setValue(L.ttl).setFontWeight('bold').setFontSize(14);

  sh.getRange(34, 1).setValue(L.t4).setFontWeight('bold').setFontSize(12);
  var stats = [
    [L.all,   '=COUNTA(' + "'" + SHEET + "'!B2:B)"],
    [L.wrote, '=COUNTIF(' + "'" + SHEET + "'!H2:H,\"<>\")"],
    [L.mail,  '=COUNTIF(' + "'" + SHEET + "'!I2:I,\"<>\")"],
    [L.gavg,  '=IFERROR(AVERAGE(' + "'" + SHEET + "'!G2:G),\"\")"],
    [L.ppl,   '=IFERROR(COUNTA(UNIQUE(FILTER(' + "'" + SHEET + "'!J2:J," + "'" + SHEET + "'!J2:J<>\"\"))),0)"],
    [L.perp,  '=IFERROR(COUNTIF(' + "'" + SHEET + "'!J2:J,\"<>\")/B39,\"\")"]
  ];
  for (var k = 0; k < stats.length; k++) {
    sh.getRange(35 + k, 1).setValue(stats[k][0]);
    sh.getRange(35 + k, 2).setFormula(stats[k][1]);
  }

  var Q2 = "'" + SHEET + "'!A:J";
  var inner = 'select J, C where J is not null and C is not null group by J, C';
  var f5 = '=IFERROR(QUERY(QUERY(' + Q2 + ',"' + inner + '",1),' +
           '"select Col1, count(Col2) group by Col1 order by count(Col2) desc label ' +
           lbl('Col1', L.rid) + ', ' + lbl('count(Col2)', L.issues) + '",1),"")';
  sh.getRange(43, 1).setValue(L.t5).setFontWeight('bold').setFontSize(12);
  sh.getRange(44, 1).setFormula(f5);

  sh.getRange('B:E').setNumberFormat('0.##');
  sh.setColumnWidth(1, 190);
  sh.getRange(38, 2).setNumberFormat('0.00');
  sh.getRange(40, 2).setNumberFormat('0.0');
  SpreadsheetApp.flush();
