איסוף המשוב לגוגל שיטס

שבע דקות של הקמה, פעם אחת. מכאן והלאה הכל אוטומטי.

מה יקרה בפועל

  • הקורא לוחץ על כוכב בסוף מדור. נפתח דף תודה שלנו - הדירוג נרשם באותו רגע, בלי שיצטרך לעשות עוד משהו.
  • בדף יש תיבת טקסט אופציונלית ותיבת סימון "אפשר לחזור אליי במייל", וכפתור חזרה לגיליון.
  • כל זה נכתב כשורה בגיליון שלך. עילום שם, אלא אם סימן את התיבה.
  • פעם בשבוע אני קורא את הגיליון ושולח לך דוח קצר: מספרים לפי מדור, מגמה, וכל ההערות במלואן.

ההקמה - שלב שלב

  1. פותחים גיליון חדש
    ב-Google Drive: חדש > Google Sheets. השם לא משנה - למשל משוב עלון שבועי.
  2. נכנסים ל-Apps Script
    בתפריט של הגיליון: הרחבות > Apps Script. נפתחת לשונית עם קובץ Code.gs וכמה שורות ריקות.
  3. מדביקים את הקוד
    מוחקים את מה שיש שם, מדביקים את כל הקוד מהתיבה למטה (יש כפתור העתקה), ושומרים - אייקון הדיסקט או Ctrl+S.
  4. מריצים פעם אחת setupSheets
    למעלה יש תיבת בחירה של פונקציה. בוחרים setupSheets ולוחצים Run. גוגל תבקש אישור הרשאה - זה החשבון שלך על הגיליון שלך, מאשרים. אחרי שנייה יופיעו בגיליון חמש לשוניות מוכנות.
  5. מפרסמים כ-Web App
    בפינה: Deploy > New deployment, ובגלגל השיניים בוחרים Web app. שם: משוב. Execute as: Me. Who has access: Anyone - זה הכרחי, אחרת קליק של קורא לא ייכתב. לוחצים Deploy.
  6. שולחים לי את הכתובת
    מתקבלת כתובת שמסתיימת ב-/exec. מעתיקים אותה ושולחים לי בסלאק. אני משתיל אותה בדף שנפתח אחרי הקליק - וזה מתחיל לעבוד.
  7. פרסום ה-CSV (רק לדוח השבועי)
    בגיליון: קובץ > שיתוף > פרסום באינטרנט, בוחרים את הלשונית נתונים ואת הפורמט CSV, ושולחים לי גם את הלינק הזה. משם אני קורא את הנתונים פעם בשבוע ושולח לך דוח מסודר - בלי שאצטרך גישה לחשבון גוגל שלך.

אם משהו נתקע באמצע - תשלח לי צילום מסך של מה שאתה רואה ואני אגיד בדיוק מה ללחוץ.

הקוד להדבקה

/**
 * GYE - העלון השבועי: איסוף דירוגי מדורים לגוגל שיטס.
 *
 * להדביק ב-Apps Script של הגיליון (תפריט: הרחבות > Apps Script), לשמור,
 * ואז: Deploy > New deployment > Web app > Execute as: Me > Who has access: Anyone.
 * את כתובת ה-exec שמתקבלת שולחים לבוט.
 *
 * הרצה חד-פעמית של setupSheets() תיצור את כל הגליונות והנוסחאות.
 * מכאן והלאה הכל מתעדכן מעצמו - אין מה לתחזק, גם אחרי שנה.
 */

var RAW = 'נתונים';
var SECTIONS = { '1': 'המאור שבה', '2': 'מתוק לנפש', '3': 'צעדני ואושעה', '4': 'צידה לדרך' };
// עמודה G נשמרת רק אם הקורא סימן בעצמו "אפשר לחזור אליי במייל". אחרת היא נשארת ריקה.
var HEADERS = ['חתימת זמן', 'גיליון', 'פרשה', 'מדור', 'דירוג', 'הערה', 'מייל לחזרה', 'מזהה'];

function doGet(e)  { return json_(record_(e.parameter || {})); }
function doPost(e) {
  var d = {};
  try { d = JSON.parse(e.postData.contents); } catch (err) { d = e.parameter || {}; }
  return json_(record_(d));
}

/** כל קליק על כוכב נרשם כשורה. הערה חופשית מתחברת לאותה שורה לפי המזהה. */
function record_(d) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = raw_();
    var id = String(d.id || Utilities.getUuid());
    var comment = String(d.c || '').trim();
    var email = String(d.e || '').trim();

    if (comment || email) {
      var n = Math.max(sh.getLastRow() - 1, 0);
      if (n) {
        var ids = sh.getRange(2, 8, n, 1).getValues();
        for (var i = ids.length - 1; i >= 0; i--) {
          if (String(ids[i][0]) === id) {
            if (comment && comment !== '-') sh.getRange(i + 2, 6).setValue(comment);
            if (email) sh.getRange(i + 2, 7).setValue(email);
            return { ok: true, updated: true, id: id };
          }
        }
      }
    }

    var rating = Number(d.r || 0);
    sh.appendRow([
      new Date(),
      Number(d.g || 0),
      String(d.p || ''),
      SECTIONS[String(d.s)] || String(d.s || ''),
      rating > 0 ? rating : '',
      comment === '-' ? '' : comment,
      email,
      id,
    ]);
    return { ok: true, id: id };
  } finally {
    lock.releaseLock();
  }
}

function raw_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(RAW);
  if (!sh) {
    sh = ss.insertSheet(RAW, 0);
    sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS])
      .setFontWeight('bold').setBackground('#dbe8f5').setFontColor('#1c4587');
    sh.setFrozenRows(1);
    sh.setRightToLeft(true);
    sh.getRange('A:A').setNumberFormat('dd/MM/yyyy HH:mm');
    sh.setColumnWidth(1, 130); sh.setColumnWidth(3, 110);
    sh.setColumnWidth(4, 130); sh.setColumnWidth(6, 380); sh.setColumnWidth(7, 200);
    sh.hideColumns(8);
  }
  return sh;
}

/** נוסחאות QUERY - הן מתרעננות לבד, כך שאין צורך לגעת בגיליון אחרי שבועות. */
function setupSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  raw_();

  var tabs = [
    { name: 'סיכום לפי מדור',
      head: ['גיליון', 'מדור', 'מספר דירוגים', 'ממוצע'],
      f: '=IFERROR(QUERY(\'' + RAW + '\'!A2:H, "select B, D, count(E), avg(E) where E > 0 group by B, D order by B desc, D", 0), "אין נתונים עדיין")' },
    { name: 'מגמה לפי מדור',
      head: [],
      f: '=IFERROR(QUERY(\'' + RAW + '\'!A2:H, "select B, avg(E) where E > 0 group by B order by B desc pivot D", 0), "אין נתונים עדיין")' },
    { name: 'הערות',
      head: ['תאריך', 'גיליון', 'מדור', 'ההערה', 'מייל לחזרה (אם ביקש)'],
      f: '=IFERROR(QUERY(\'' + RAW + '\'!A2:H, "select A, B, D, F, G where F is not null and F <> \'\' order by A desc", 0), "אין הערות עדיין")' },
    { name: 'מבקשים חזרה במייל',
      head: ['תאריך', 'גיליון', 'מדור', 'דירוג', 'ההערה', 'מייל'],
      f: '=IFERROR(QUERY(\'' + RAW + '\'!A2:H, "select A, B, D, E, F, G where G is not null and G <> \'\' order by A desc", 0), "אין בקשות עדיין")' },
    { name: 'פילוח דירוגים',
      head: ['דירוג', 'כמה פעמים'],
      f: '=IFERROR(QUERY(\'' + RAW + '\'!A2:H, "select E, count(E) where E > 0 group by E order by E desc", 0), "אין נתונים עדיין")' },
  ];

  tabs.forEach(function (t) {
    var sh = ss.getSheetByName(t.name) || ss.insertSheet(t.name);
    sh.clear();
    sh.setRightToLeft(true);
    var row = 1;
    if (t.head.length) {
      sh.getRange(1, 1, 1, t.head.length).setValues([t.head])
        .setFontWeight('bold').setBackground('#dbe8f5').setFontColor('#1c4587');
      sh.setFrozenRows(1);
      row = 2;
    }
    sh.getRange(row, 1).setFormula(t.f);
    sh.setColumnWidth(1, 130);
    if (t.name === 'הערות') { sh.setColumnWidth(4, 480); sh.setColumnWidth(5, 200); }
    if (t.name === 'מבקשים חזרה במייל') { sh.setColumnWidth(5, 380); sh.setColumnWidth(6, 200); }
    if (t.name === 'סיכום לפי מדור') sh.getRange('D2:D').setNumberFormat('0.00');
    if (t.name === 'מגמה לפי מדור') sh.getRange('B2:F').setNumberFormat('0.00');
  });

  ss.setActiveSheet(ss.getSheetByName('סיכום לפי מדור'));
  return 'נוצרו ' + (tabs.length + 1) + ' גליונות';
}

איך הגיליון יישאר ברור גם אחרי חצי שנה

  • נתונים - כל קליק שורה אחת: תאריך, גיליון, פרשה, מדור, דירוג, הערה, ומייל - רק אם הקורא סימן שאפשר לחזור אליו. זו הטבלה הגולמית, ממנה הכל נגזר.
  • סיכום לפי מדור - לכל גיליון ולכל מדור: כמה דירוגים וממוצע. הגיליון החדש נכנס למעלה, אז אין גלילה של חצי שנה.
  • מגמה לפי מדור - טבלה צולבת: שורה לכל גיליון, עמודה לכל מדור, בתוכה הממוצע. זו הלשונית שבה רואים בעין אם מדור מתחזק או נחלש.
  • הערות - רק השורות שיש בהן טקסט, מהחדש לישן, עם המדור והמייל לצידן.
  • מבקשים חזרה במייל - רק מי שסימן במפורש. זו רשימת המענה שלך.
  • פילוח דירוגים - כמה 5, כמה 4 וכן הלאה - על כל הזמנים.

כל הלשוניות חוץ מנתונים בנויות על נוסחאות QUERY שמתעדכנות לבד. אין מה לגרור, אין מה לרענן, ואף שורה לא נמחקת - הגיליון פשוט ממשיך לגדול למטה.
